Google 스프레드시트 Apps Script 연동

구글 스프레드시트(Apps Script Web App)와 직접 양방향 동기화할 수 있는 연동 URL을 등록하세요.

[배포] -> [새 배포] -> [유형: 웹 앱] -> [액세스 권한: 모든 사용자]로 배포한 URL을 입력합니다.
📋 초간단 Google Apps Script 코드 예시 (복사해서 붙여넣기) 
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  // 첫 열: 학생, 두번째 열: 역할
  return ContentService.createTextOutput(JSON.stringify({data: data}))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var params = JSON.parse(e.postData.contents);
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  sheet.clear();
  sheet.appendRow(["번호", "학생 이름", "배정 역할", "배정일시"]);
  params.results.forEach(function(item, idx) {
    sheet.appendRow([idx + 1, item.student, item.role, params.date]);
  });
  return ContentService.createTextOutput(JSON.stringify({status: "success"}))
    .setMimeType(ContentService.MimeType.JSON);
}
